## mock fallback 设计 (LLM 失败时)

```ts
function mockExtractWords(text: string, maxWords = 15): CustomWord[] {
  // 1. 简单分词 (英文)
  const words = text.toLowerCase().match(/[a-z]+/g) || []
  // 2. 过滤常见词 (a, the, is, ...)
  const stopwords = new Set(['a', 'an', 'the', 'is', 'are', 'was', 'were', 'be', 'been', ...])
  const filtered = words.filter(w => !stopwords.has(w) && w.length >= 4)
  // 3. 词频
  const freq = new Map<string, number>()
  filtered.forEach(w => freq.set(w, (freq.get(w) || 0) + 1))
  // 4. 排序取 top N
  const top = [...freq.entries()].sort((a, b) => b[1] - a[1]).slice(0, maxWords)
  // 5. 假释义 + 假例句
  return top.map(([w]) => ({
    word: w,
    translation: `${w} (释义占位)`,
    example: `Example with ${w}.`,
    difficulty: 'B1',
  }))
}
```
